import pandas as pd
import dash
from dash import dcc, html
import plotly.express as px

# ==============================
# LEITURA DO CSV
# ==============================
df = pd.read_csv("ecommerce_estatistica.csv")

# ==============================
# GRÁFICOS
# ==============================

fig_scatter = px.scatter(
    df,
    x="Preço",
    y="Nota",
    color="Gênero",
    title="Preço vs Nota dos Produtos",
    hover_data=["Título", "Marca"]
)

fig_bar = px.bar(
    df.groupby("Marca", as_index=False)["Nota"].mean(),
    x="Marca",
    y="Nota",
    title="Média de Nota por Marca"
)

fig_hist = px.histogram(
    df,
    x="Preço",
    nbins=30,
    title="Distribuição de Preços"
)

fig_vendas = px.bar(
    df.groupby("Gênero", as_index=False)["Qtd_Vendidos"].sum(),
    x="Gênero",
    y="Qtd_Vendidos",
    title="Quantidade Vendida por Gênero"
)

# ==============================
# APP DASH
# ==============================
app = dash.Dash(__name__)
app.title = "Dashboard E-commerce"

app.layout = html.Div([
    html.H1("Dashboard E-commerce", style={"textAlign": "center"}),

    dcc.Graph(figure=fig_scatter),
    dcc.Graph(figure=fig_bar),
    dcc.Graph(figure=fig_hist),
    dcc.Graph(figure=fig_vendas)
])

# ==============================
# EXECUÇÃO
# ==============================
if __name__ == "__main__":
    app.run_server(debug=True)
